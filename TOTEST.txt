# Supprimer la licence (reset pour retester)
node -e "const {PrismaClient}=require('@prisma/client');const p=new PrismaClient();p.licenseHistory.deleteMany().then(()=>p.instanceConfig.updateMany({data:{licenseKey:null,licenseId:null,company:null,email:null,licenseExpiresAt:null,contentKey:null,renewalInProgress:false}})).then(()=>{console.log('ok');p.\$disconnect()})"

# Générer une nouvelle licence (1 mois)
cd lms-license-cli && node index.js issue --company "danjecorp" --email "danje@danjecorp.com" --expires $(date -d '+1 month' '+%Y-%m-%d')

# Voir l'état de la licence en base
node -e "const {PrismaClient}=require('@prisma/client');const p=new PrismaClient();p.instanceConfig.findFirst({select:{licenseId:true,company:true,email:true,licenseExpiresAt:true,renewalInProgress:true}}).then(r=>{console.log(r);p.\$disconnect()})"

# Lister les clients CLI
cd lms-license-cli && node index.js list
